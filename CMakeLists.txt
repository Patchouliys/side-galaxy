cmake_minimum_required(VERSION 3.20)
project(side_galaxy_core VERSION 0.1.0 LANGUAGES CXX)

find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(SG_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(SG_SQLITE_TARGET SQLite::SQLite3)
endif()
add_library(side_galaxy_core SHARED native/core.cpp native/host.cpp)
target_compile_features(side_galaxy_core PRIVATE cxx_std_20)
target_compile_definitions(side_galaxy_core PRIVATE SG_CORE_BUILD)
target_include_directories(side_galaxy_core PRIVATE native/vendor PUBLIC native)
target_link_libraries(side_galaxy_core PRIVATE ${SG_SQLITE_TARGET})
set_target_properties(side_galaxy_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(side_galaxy_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(side_galaxy_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS side_galaxy_core LIBRARY DESTINATION side_galaxy/_native RUNTIME DESTINATION side_galaxy/_native)

include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(side_galaxy_core_test native/core_test.cpp)
  target_compile_features(side_galaxy_core_test PRIVATE cxx_std_20)
  target_include_directories(side_galaxy_core_test PRIVATE native/vendor)
  target_link_libraries(side_galaxy_core_test PRIVATE side_galaxy_core ${SG_SQLITE_TARGET} Threads::Threads)
  add_test(NAME side_galaxy_core_transactions COMMAND side_galaxy_core_test)
endif()
